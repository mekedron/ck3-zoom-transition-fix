# Overrides game/gfx/map/map_object_data/game_object_layers.txt
#
# Vanilla drops activities, buildings and units all on zoom step 9, the same
# step that unloads the three tree layers, the coast foam and the mountain
# env effects, hides the fort icons, swaps the map names and starts the realm
# colour overlay. Everything is torn down in one frame.
#
# These three move off that step:
#
#   layer             vanilla  here   camera height
#   activities_layer        9     7   295
#   building_layer          9    10   453
#   unit_layer              9    10   453
#
# Holdings go on step 10, where the realm colour overlay starts filling (this
# mod moves the start there from 9), so the castles and towns leave as the
# colours come rather than two steps before them. The overlay is faint for
# its first two steps and clearly visible from 12, but a layer pushed higher
# stays visible over a wider zoom range, and the visible map area grows fast
# as you pull back: an earlier version of this mod kept buildings to step 11
# and it cost frame rate exactly while zooming in towards county and duchy
# level. 10 is the compromise; 12 is what the AGOT patch uses, on a map with
# fewer holdings. Change the number below if you want to try it.

layer={
	name="activities_layer"
	fade_in=0
	fade_out=7
	category=""
	masks=""
	visibility_tags=""
}
layer={
	name="building_layer"
	fade_in=0
	fade_out=10
	category=""
	masks=""
	visibility_tags=""
}
layer={
	name="unit_layer"
	fade_in=0
	fade_out=10
	category=""
	masks=""
	visibility_tags=""
}
